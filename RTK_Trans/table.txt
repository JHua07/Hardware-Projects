# ======================================================
# GNSS通道跟踪状态字解析注释（32位十六进制状态字）
# 数据来源：表7-20、卫星系统映射表、信号类型表
# ======================================================

# ----------------------------
# 表7-20 通道跟踪状态（N0-N3）
# ----------------------------
# N0 (bit 0-3): 保留字段
#   0x0000000F 无实际意义

# N1 (bit 4-7): SV通道号
#   0x000000F0
#   取值范围：0-n (0=第一个通道，n=接收机最大通道数)

# N2 (bit 8-11): 载波相位有效性
#   0x00000F00
#   bit8: 载波相位有效标志 (0=无效, 1=有效)
#   bit9-11: 保留

# N3 (bit 12-14): 伪距及其他标志
#   0x0000F000
#   bit12: 伪距有效标志 (0=无效, 1=有效)
#   bit13-14: 保留

# ----------------------------
# 卫星系统映射（N4-N6）
# ----------------------------
# N4 (bit 16-19): 卫星系统类型
#   0x000F0000
#   取值：
#     0 = GPS (G)
#     1 = GLONASS (R)
#     2 = SBAS (S)
#     3 = Galileo (E)
#     4 = BDS (C)
#     5 = QZSS (J)
#     6-7 = 保留

# N5 (bit 20-23): 信号类型主标识
#   0x00F00000
#   具体取值依赖卫星系统（见下方信号类型表）

# N6 (bit 24-27): 扩展标志
#   0x0F000000
#   bit26: L2C标志位 (0=L2P(Y), 1=L2C)
#   其他位保留

# ----------------------------
# 信号类型详细映射（按卫星系统）
# ----------------------------
# GPS信号类型 (N5取值):
#   0 = L1 C/A  -> RINEX: C1C L1C D1C S1C
#   3 = L1C (pilot+data) -> C1X L1X D1X S1X
#   6 = L5 data -> C5I L5I D5I S5I
#   9 = L2P(Y)  -> C2P L2P D2P S2P
#   14 = L5 pilot -> C5Q L5Q D5Q S5Q
#   17 = L2C(L) -> C2L L2L D2L S2L

# BDS信号类型:
#   0 = B1I -> C1I L1I D1I S1I
#   4 = B1Q -> C1Q L1Q D1Q S1Q
#   8 = B1C(Pilot) -> C1P L1P D1P S1P
#   12 = B2a(Pilot) -> C7Q L7Q D7Q S7Q
#   17 = B2I -> C2I L2I D2I S2I
#   21 = B3I -> C6I L6I D6I S6I
#   23 = B1C(Data) -> C1D L1D D1D S1D
#   28 = B2a(Data) -> C7D L7D D7D S7D

# GLONASS信号类型:
#   0 = L1 C/A -> C1C L1C D1C S1C
#   5 = L2 C/A -> C2C L2C D2C S2C
#   6 = G3I -> C3I L3I D3I S3I
#   7 = G3Q -> C3Q L3Q D3Q S3Q

# Galileo信号类型:
#   1 = E1B -> C1B L1B D1B S1B
#   2 = E1C -> C1C L1C D1C S1C
#   12 = E5a -> C5Q L5Q D5Q S5Q
#   17 = E5b -> C7Q L7Q D7Q S7Q
#   18 = E6B -> C6B L6B D6B S6B
#   22 = E6C -> C6C L6C D6C S6C

# QZSS信号类型:
#   0 = L1 C/A -> C1C L1C D1C S1C
#   6 = L5 data -> C5I L5I D5I S5I
#   14 = L5 pilot -> C5Q L5Q D5Q S5Q
#   17 = L2C(L) -> C2L L2L D2L S2L

# SBAS信号类型:
#   0 = L1 C/A -> C1C L1C D1C S1C
#   6 = L5 -> C5I L5I D5I S5I

# ----------------------------
# 有效性检查标志位
# ----------------------------
# 载波相位有效: (status_word & 0x00000100) >> 8
# 伪距有效: (status_word & 0x00001000) >> 12
# L2C标志: (status_word & 0x04000000) >> 26
